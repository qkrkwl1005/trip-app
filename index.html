(function initPullToRefresh(){ let startY=0,startX=0,pulling=false; window.addEventListener('touchstart',e=>{ // 지도 탭이거나 지도 영역, 상세 시트, 오버레이, 2손가락(핀치 줌) 터치 시 새로고침 완전 차단 if( e.touches.length!==1|| document.body.classList.contains('show-map')|| e.target.closest('#map-wrap,#sheet,#myMapsOverlay,input,textarea,select') ){ pulling=false; return; } // 상단 헤더(화면 맨 위 70px 이내)에서 당겼을 때만 새로고침 허용 const sc=e.target.closest('.tab-content'); if((sc&&sc.scrollTop>0)||e.touches[0].clientY>70){ pulling=false; return; } startY=e.touches[0].clientY; startX=e.touches[0].clientX; pulling=true; },{passive:true}); window.addEventListener('touchmove',e=>{ if(!pulling||e.touches.length!==1){ pulling=false; $('ptr').classList.remove('show'); return; } const dy=e.touches[0].clientY-startY; const dx=Math.abs(e.touches[0].clientX-startX); if(dy>130&&dx<40)$('ptr').classList.add('show'); else $('ptr').classList.remove('show'); },{passive:true}); window.addEventListener('touchend',()=>{ if(pulling&&$('ptr').classList.contains('show')){ $('ptr').classList.remove('show'); location.reload(); } pulling=false; },{passive:true}); })();
